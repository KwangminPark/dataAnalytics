기본 실습 1 · 작은 예제로 추출과 회전 익히기

9문항에서 평행분석으로 요인 수를 살피고 세 요인의 PAF를 회전 전/varimax 후 비교합니다. 적재량은 문항과 요인 관계, 공통성은 선택한 요인으로 설명되는 문항 분산입니다. 회전은 새 고객자료를 만드는 과정이 아니라 축을 읽기 쉽게 바꾸는 과정입니다.

기본 실습 2 · 실제 설문의 열을 감사

survey-audit.csv에서 열 수·범위·소수값을 확인합니다. 이름으로 문항을 선택합니다. 분석 목적상 역할이 다른 설명영역과 결과영역을 나누되 요인분석으로 인과 방향이 정해지는 것은 아닙니다. 코드의 문항 제외는 교재 결과 재현을 위한 선택입니다. 연구 정답으로 간주하지 말고 어떤 개념을 잃는지 기록하세요.

기본 실습 3 · 교재의 장말 결과 재현

교재의 단계별 선택을 stage-iv_27→iv_24→iv_21→survey-iv-loadings(17문항), stage-dv_17→dv_14→dv_10→survey-dv-varimax(7문항) 순서로 비교합니다. 가격·메뉴·신뢰·명성·내만족을 제외하면 어떤 의미를 잃는지 적습니다. 이후 설명 문항 17개의 5요인 점수와 외만족 4문항+재방문 3문항의 varimax 점수로 회귀를 실행합니다. 어떤 열이 만족/재방문인지 적재표로 찾고 평균 문항과 방향을 맞춥니다. PA1을 무조건 만족으로 이름 바꾸지 말고, 음의 만족 계수가 재현되는지 확인합니다. 최초 결과영역 17문항 모형은 재방문2의 공통성이 약 1.018로 1을 넘어 Heywood 경고가 납니다. 이 중간 모형을 정상적인 점수로 채택하지 말고 과도한 요인 수·문항 중복·추출 가정을 점검할 대상으로 표시하세요.

기본 실습 4 · 음의 계수의 출처를 좁히기

varimax 점수의 만족-재방문 단순상관과 다른 요인점수를 넣은 조건부 계수를 비교합니다. 이어 같은 7문항에 oblimin 회전과 문항평균을 사용합니다. 회전/점수 정의만 바꿨을 때 관계가 얼마나 달라지는지 읽습니다. 점수 방식이 달라 계수의 절대 크기를 효과 크기처럼 직접 비교하지 말고 방향·상관·구간을 중심으로 봅니다.

기본 실습 5 · 전략으로 넘어가기 전 보류할 것

varimax는 요인 간 상관을 0으로 두는 모형입니다. 후속 분석에서 만족과 재방문의 관계를 묻는데 이 구성 방식이 적절한지 논의합니다. 음의 조건부 계수 자체가 입력 오류의 증명은 아닙니다. 억제효과, 점수 구성, 원문항·역코딩, 표본 특성을 확인합니다. 원문항 확인 없이 재방문3을 역문항이라고 임의 반전하지 않습니다.

```{r}
# 2. 평행분석, 주축요인법(PAF), 회전 전후 비교
set.seed(20261002)
picture("parallel-analysis"); pa <- psych::fa.parallel(x,fa="fa",fm="pa",n.iter=100,plot=TRUE); dev.off()
cat("Parallel analysis suggested factors:",pa$nfact,"
")
f0 <- psych::fa(x,nfactors=3,fm="pa",rotate="none")
f <- psych::fa(x,nfactors=3,fm="pa",rotate="varimax")
fo <- psych::fa(x,nfactors=3,fm="pa",rotate="oblimin")
export(data.frame(item=names(x),unclass(f$loadings),h2=f$communality,u2=f$uniqueness),"factor-loadings")
export(data.frame(item=names(x),unclass(f0$loadings)),"unrotated")
print(fo$Phi)
```

```{r}
#| label: analysis-5
# 5. 9.7: 실제 설문. 416행/45열이며 성별 1열 + 문항 44열입니다.
d <- read_data("ch9_data1.csv")
stopifnot(nrow(d)==416,ncol(d)==45,!anyNA(d))
export(data.frame(item=names(d),min=sapply(d,min),max=sapply(d,max),noninteger=sapply(d,function(x) sum(abs(x-round(x))>1e-8))),"survey-audit")
# 장말의 단계별 문항 선택을 재현하되, 선택 자체를 정답으로 취급하지 않습니다.
initial_iv <- c(paste0("분위기",1:5),paste0("접근성",1:4),paste0("편의성",1:4),paste0("친절도",1:5),paste0("맛",1:3),paste0("가격",1:3),paste0("메뉴",1:3))
initial_dv <- c(paste0("명성",1:4),paste0("외만족",1:4),paste0("내만족",1:3),paste0("신뢰",1:3),paste0("재방문",1:3))
stages <- list(iv_27=list(initial_iv,7),iv_24=list(setdiff(initial_iv,paste0("가격",1:3)),6),iv_21=list(setdiff(initial_iv,c(paste0("가격",1:3),paste0("메뉴",1:3))),5),dv_17=list(initial_dv,4),dv_14=list(setdiff(initial_dv,paste0("신뢰",1:3)),3),dv_10=list(c(paste0("외만족",1:4),paste0("내만족",1:3),paste0("재방문",1:3)),2))
for(stage in names(stages)) {
 spec <- stages[[stage]]
 fit_stage <- psych::fa(d[spec[[1]]],nfactors=spec[[2]],fm="pa",rotate="varimax")
 export(data.frame(item=spec[[1]],unclass(fit_stage$loadings),h2=fit_stage$communality),paste0("stage-",stage))
 if(any(fit_stage$communality>1)) cat("DIAGNOSTIC:",stage,"has communality > 1; this is an improper exploratory solution, not an accepted score model.\n")
}
# 위치 번호가 아니라 문항 이름으로 선택하여 잘못된 열 삭제를 방지합니다.
iv_names <- c(paste0("분위기",2:5),paste0("접근성",1:3),paste0("편의성",2:4),paste0("친절도",2:5),paste0("맛",1:3))
iv <- d[iv_names]
fi <- psych::fa(iv,nfactors=5,fm="pa",rotate="varimax")
si <- psych::factor.scores(iv,fi,method="regression")$scores
# 열 이름을 자동으로 경영 개념으로 바꾸지 않습니다. 적재표를 함께 저장합니다.
export(data.frame(item=iv_names,unclass(fi$loadings)),"survey-iv-loadings")
dv <- d[c(paste0("외만족",1:4),paste0("재방문",1:3))]
# 각 문항 묶음과 가장 관련된 요인을 찾고 방향을 맞춥니다.
score_pair <- function(rotation) {
 fit <- psych::fa(dv,nfactors=2,fm="pa",rotate=rotation)
 scores <- psych::factor.scores(dv,fit,method="regression")$scores
 L <- unclass(fit$loadings)
 sat_col <- which.max(colMeans(abs(L[1:4,,drop=FALSE])))
 rev_col <- which.max(colMeans(abs(L[5:7,,drop=FALSE])))
 stopifnot(sat_col!=rev_col)
 sat <- scores[,sat_col]*sign(cor(scores[,sat_col],rowMeans(dv[,1:4])))
 rev <- scores[,rev_col]*sign(cor(scores[,rev_col],rowMeans(dv[,5:7])))
 export(data.frame(item=names(dv),L),paste0("survey-dv-",rotation))
 if(!is.null(fit$Phi)) print(fit$Phi)
 data.frame(satisfaction=sat,revisit=rev)
}
v <- score_pair("varimax"); o <- score_pair("oblimin")
means <- data.frame(satisfaction=rowMeans(dv[,1:4]),revisit=rowMeans(dv[,5:7]))
compare <- function(pair,label) {
 df <- cbind(pair,as.data.frame(si))
 model <- lm(revisit~.,data=df)
 coefrow <- coef_table(model); print(coefrow)
 r <- coefrow[coefrow$term=="satisfaction",]
 data.frame(scores=label,marginal_correlation=cor(pair$satisfaction,pair$revisit),adjusted_satisfaction=r$estimate,low=r$low,high=r$high,p=r$p)
}
export(rbind(compare(v,"varimax"),compare(o,"oblimin"),compare(means,"item-means")),"score-method-comparison")
picture("score-construction",1200,650); par(mfrow=c(1,2))
plot(v$satisfaction,v$revisit,pch=16,col="#236b6655",xlab="Satisfaction score",ylab="Revisit score",main="Varimax: marginal relation"); abline(lm(revisit~satisfaction,v),col=2,lwd=2)
plot(o$satisfaction,o$revisit,pch=16,col="#236b6655",xlab="Satisfaction score",ylab="Revisit score",main="Oblimin: marginal relation"); abline(lm(revisit~satisfaction,o),col=2,lwd=2)
dev.off()
# 각 경로와 a*b의 점추정: 유의한 간접효과라고 부르려면 별도 불확실성 평가가 필요.
path_data <- cbind(v,as.data.frame(si))
ma <- lm(satisfaction~.,data=path_data[setdiff(names(path_data),"revisit")])
mb <- lm(revisit~.,data=path_data)
a <- coef(ma)[colnames(si)]; b <- coef(mb)["satisfaction"]
export(data.frame(predictor=names(a),a=as.numeric(a),b=as.numeric(b),ab=as.numeric(a*b)),"survey-path-products")
cat("Negative conditional coefficients are not proof of a data error. Compare score definitions, marginal/partial relations and item wording before drawing conclusions.
")
```

문항 감사표, 설명영역·결과영역 적재표, 세 점수 방식의 상관/계수 비교, 음의 계수에 대한 가능한 설명과 아직 확인하지 못한 조건을 제출합니다. “유의하니 매장 이전” 같은 실행 불가능한 제안 대신 운영자가 실제 바꿀 수 있는 항목을 제안합니다. 질문→자료→방법→근거→실행/보류 판단을 연결합니다. 공식 제출은 집현캠퍼스 안내를 따릅니다. 마감일·배점은 이 자료에서 정하지 않습니다. 스스로 확인: 음수를 없애려고 자료를 바꾸지 않았는가? 34와 44문항의 차이를 확인했는가?
