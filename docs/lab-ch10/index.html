## 4. 교수 제공 혼합형 자료: 연속값 + 범주형, Gower 거리로 비교

```{r}
#| label: analysis-4
# 4. 교수 제공 혼합형 자료: 연속값 + 범주형, Gower 거리로 비교
mixed <- read_data("ch10_data1.csv")
stopifnot(nrow(mixed)==500,!anyNA(mixed),!anyDuplicated(mixed$id))
mixed$day_pref <- factor(mixed$day_pref)
mixed$menu_pref <- factor(mixed$menu_pref)
gower <- cluster::daisy(mixed[c("visit_freq","avg_spend","day_pref","menu_pref")],metric="gower")
gm <- cutree(hclust(gower,method="complete"),k=3)
export(aggregate(mixed[c("visit_freq","avg_spend")],list(cluster=gm),mean),"mixed-profile")
export(as.data.frame(table(cluster=gm,day=mixed$day_pref)),"mixed-day")
export(as.data.frame(table(cluster=gm,menu=mixed$menu_pref)),"mixed-menu")
```
